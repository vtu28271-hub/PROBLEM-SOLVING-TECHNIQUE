import java.util.*;

public class FrequencyQueries {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int q = sc.nextInt();

        HashMap<Integer, Integer> freq = new HashMap<>();
        HashMap<Integer, Integer> count = new HashMap<>();

        for (int i = 0; i < q; i++) {
            int type = sc.nextInt();
            int x = sc.nextInt();

            if (type == 1) {
                int old = freq.getOrDefault(x, 0);

                if (old > 0) {
                    count.put(old, count.get(old) - 1);
                }

                freq.put(x, old + 1);

                count.put(old + 1,
                    count.getOrDefault(old + 1, 0) + 1);
            }

            else if (type == 2) {
                int old = freq.getOrDefault(x, 0);

                if (old > 0) {
                    count.put(old, count.get(old) - 1);

                    freq.put(x, old - 1);

                    if (old - 1 > 0) {
                        count.put(old - 1,
                            count.getOrDefault(old - 1, 0) + 1);
                    }
                }
            }

            else if (type == 3) {
                System.out.println(
                    count.getOrDefault(x, 0) > 0 ? 1 : 0
                );
            }
        }
    }
}
