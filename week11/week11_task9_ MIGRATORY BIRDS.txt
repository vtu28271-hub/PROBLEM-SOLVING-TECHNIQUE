import java.util.*;

public class MIGRATORY BIRDS {
    public static void main(String[] args) {
        int[] arr = {1, 4, 4, 4, 5, 3};
        int[] count = new int[6];

        for (int bird : arr) {
            count[bird]++;
        }

        int max = 0;
        int answer = 0;
        for (int i = 1; i <= 5; i++) {
            if (count[i] > max) {
                max = count[i];
                answer = i;
            }
        }

        System.out.println(answer);
    }
}
